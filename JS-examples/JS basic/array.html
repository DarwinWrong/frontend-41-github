        Завдання: Перетворити масив об'єктів на об'єкт, де ключі — це імена, а значення — це віки.

        let people = [
        { name: "Alice", age: 25 },
        { name: "Bob", age: 30 },
        { name: "Charlie", age: 35 }
        ];

        let ages = people.reduce((acc, person) => {
        acc[person.name] = person.age;
        return acc;
        }, {});

        console.log(ages); // { Alice: 25, Bob: 30, Charlie: 35 }

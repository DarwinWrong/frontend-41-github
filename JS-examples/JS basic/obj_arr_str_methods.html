Методи, які працюють виключно з об'єктами (Object)

Метод Опис
Object.keys(obj) Повертає масив з ключами об'єкта. 
const obj = { a: 1, b: 2, c: 3 };
  console.log(Object.keys(obj)); // ["a", "b", "c"]
Object.values(obj) Повертає масив значень об'єкта. 
const obj = { a: 1, b: 2, c: 3 };
  console.log(Object.values(obj)); // [1, 2, 3]
Object.entries(obj) Повертає масив пар [ключ, значення]. 
const obj = { a: 1, b: 2 };
  console.log(Object.entries(obj)); // [["a", 1], ["b", 2]]
Object.fromEntries(arr) Створює об'єкт із масиву пар [ключ, значення]. 
const entries = [["a", 1], ["b", 2]];
  console.log(Object.fromEntries(entries)); // { a: 1, b: 2 }
Object.assign(target, ...sources) Копіює властивості з одного або кількох об'єктів у target. 
const target = { a: 1 };
  const source = { b: 2 };
  Object.assign(target, source);
  console.log(target); // { a: 1, b: 2 }
Object.freeze(obj) Заморожує об'єкт (не можна змінювати значення). 
const obj = { a: 1 };
  Object.freeze(obj);
  obj.a = 2; // Не зміниться
  console.log(obj.a); // 1
Object.seal(obj) Запобігає додаванню/видаленню властивостей (але змінювати існуючі можна). 
const obj = { a: 1 };
  Object.seal(obj);
  obj.a = 2; // Дозволено
  obj.b = 3; // Не додасться
  console.log(obj); // { a: 2 }

Методи, які працюють виключно з масивами (Array)

Метод Опис
push(el) Додає елемент у кінець масиву. 
const arr = [1, 2, 3];
      arr.push(4);
      console.log(arr); // [1, 2, 3, 4]
pop() Видаляє останній елемент масиву. 
const arr = [1, 2, 3];
      arr.pop();
      console.log(arr); // [1, 2]
shift() Видаляє перший елемент масиву. 
const arr = [1, 2, 3];
      arr.shift();
      console.log(arr); // [2, 3]
unshift(el) Додає елемент на початок масиву. 
const arr = [1, 2, 3];
      arr.unshift(0);
      console.log(arr); // [0, 1, 2, 3]
splice(start, deleteCount, ...items) Видаляє або додає елементи у масив. 
const arr = [1, 2, 3, 4];
      arr.splice(1, 2, 5, 6);
      console.log(arr); // [1, 5, 6, 4]
slice(start, end) Повертає новий масив із частиною елементів. 
const arr = [1, 2, 3, 4];
      const newArr = arr.slice(1, 3);
      console.log(newArr); // [2, 3]
concat(arr) Об'єднує масиви. 
const arr1 = [1, 2];
      const arr2 = [3, 4];
      const newArr = arr1.concat(arr2);
      console.log(newArr); // [1, 2, 3, 4]
indexOf(el) Знаходить індекс першого входження елемента. 
const arr = [1, 2, 3];
      console.log(arr.indexOf(2)); // 1
lastIndexOf(el) Знаходить індекс останнього входження елемента. 
const arr = [1, 2, 3, 2];
      console.log(arr.lastIndexOf(2)); // 3
includes(el) Перевіряє, чи є елемент у масиві. 
const arr = [1, 2, 3];
      console.log(arr.includes(2)); // true
find(fn) Повертає перший елемент, що відповідає умові. 
const arr = [1, 2, 3, 4];
      const found = arr.find(x => x > 2);
      console.log(found); // 3
findIndex(fn) Повертає індекс першого елемента, що відповідає умові. 
const arr = [1, 2, 3, 4];
      const index = arr.findIndex(x => x > 2);
      console.log(index); // 2
map(fn) Створює новий масив, перетворюючи кожен елемент. 
const arr = [1, 2, 3];
      const newArr = arr.map(x => x * 2);
      console.log(newArr); // [2, 4, 6]
filter(fn) Повертає масив із елементами, що відповідають умові. 
const arr = [1, 2, 3, 4];
      const filtered = arr.filter(x => x % 2 === 0);
      console.log(filtered); // [2, 4]
reduce(fn, init) Згортає масив до одного значення. 
const arr = [1, 2, 3, 4];
      const sum = arr.reduce((acc, curr) => acc + curr, 0);
      console.log(sum); // 10
sort(fn) Сортує масив (за замовчуванням як рядки). 
const arr = [4, 1, 3, 2];
      arr.sort();
      console.log(arr); // [1, 2, 3, 4]
reverse() Перевертає масив. 
const arr = [1, 2, 3];
      arr.reverse();
      console.log(arr); // [3, 2, 1]
forEach(fn) Виконує функцію для кожного елемента. 
const arr = [1, 2, 3];
      arr.forEach(x => console.log(x)); // 1, 2, 3

Методи, які працюють виключно з рядками (String)

Метод Опис
charAt(index) Повертає символ за індексом. 
const str = "Hello";
      console.log(str.charAt(1)); // "e"
charCodeAt(index) Повертає Unicode код символу. 
const str = "Hello";
      console.log(str.charCodeAt(1)); // 101
slice(start, end) Вирізає підрядок. 
const str = "Hello";
      console.log(str.slice(1, 4)); // "ell"
substring(start, end) Альтернативний варіант slice(). 
const str = "Hello";
      console.log(str.substring(1, 4)); // "ell"
substr(start, length) Застарілий метод, використовуйте slice(). 
const str = "Hello";
      console.log(str.substr(1, 3)); // "ell"
indexOf(str) Знаходить індекс першого входження підрядка. 
const str = "Hello";
      console.log(str.indexOf("e")); // 1
lastIndexOf(str) Знаходить індекс останнього входження підрядка. 
const str = "Hello";
      console.log(str.lastIndexOf("l")); // 3
includes(str) Перевіряє, чи містить рядок підрядок. 
const str = "Hello";
      console.log(str.includes("e")); // true
startsWith(str) Перевіряє, чи починається рядок із підрядка. 
const str = "Hello";
      console.log(str.startsWith("He")); // true
endsWith(str) Перевіряє, чи закінчується рядок підрядком. 
const str = "Hello";
      console.log(str.endsWith("lo")); // true
toUpperCase() Перетворює у верхній регістр. 
const str = "hello";
      console.log(str.toUpperCase()); // "HELLO"
toLowerCase() Перетворює у нижній регістр. 
const str = "HELLO";
      console.log(str.toLowerCase()); // "hello"
trim() Видаляє пробіли на початку і в кінці. 
const str = "  Hello  ";
      console.log(str.trim()); // "Hello"
replace(str, newStr) Замінює підрядок. 
const str = "Hello";
      console.log(str.replace("e", "a")); // "Hallo"
split(delimiter) Розбиває рядок на масив за роздільником. 
const str = "Hello World";
      console.log(str.split(" ")); // ["Hello", "World"]

Методи, які можуть працювати з різними типами

Метод Типи Опис
forEach(fn) Масив, Map, Set Виконує функцію для кожного елемента. 
const arr = [1, 2, 3];
      arr.forEach(num => console.log(num)); // 1, 2, 3
map(fn) Масив Створює новий масив, перетворюючи елементи. 
const arr = [1, 2, 3];
      console.log(arr.map(num => num * 2)); // [2, 4, 6]
filter(fn) Масив Повертає масив з елементами, що відповідають умові. 
const arr = [1, 2, 3, 4];
      console.log(arr.filter(num => num > 2)); // [3, 4]
reduce(fn, init) Масив Згортає масив до одного значення. 
const arr = [1, 2, 3];
      console.log(arr.reduce((acc, num) => acc + num, 0)); // 6
indexOf(el) Масив, рядок Шукає елемент або підрядок і повертає індекс. 
const arr = [1, 2, 3];
      console.log(arr.indexOf(2)); // 1
lastIndexOf(el) Масив, рядок Шукає останнє входження елемента або підрядка. 
const arr = [1, 2, 3, 2];
      console.log(arr.lastIndexOf(2)); // 3
includes(el) Масив, рядок Перевіряє, чи міститься елемент або підрядок. 
const arr = [1, 2, 3];
      console.log(arr.includes(2)); // true
sort(fn) Масив Сортує масив. 
const arr = [3, 1, 2];
      console.log(arr.sort()); // [1, 2, 3]
reverse() Масив Перевертає масив. 
const arr = [1, 2, 3];
      console.log(arr.reverse()); // [3, 2, 1]
